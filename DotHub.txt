local function CreatePage(tabName)
    local PageScroll = Instance.new("ScrollingFrame")
    PageScroll.Size = UDim2.new(1, -10, 1, -10)
    PageScroll.Position = UDim2.new(0, 5, 0, 5)
    PageScroll.BackgroundTransparency = 1
    PageScroll.CanvasSize = UDim2.new(0, 0, 0, 0)
    PageScroll.ScrollBarThickness = 3
    PageScroll.Visible = false
    PageScroll.Parent = ContentContainer

    local PageList = Instance.new("UIListLayout")
    PageList.SortOrder = Enum.SortOrder.LayoutOrder
    PageList.Padding = UDim.new(0, 8)
    PageList.Parent = PageScroll

    PageList:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        PageScroll.CanvasSize = UDim2.new(0, 0, 0, PageList.AbsoluteContentSize.Y + 10)
    end)

    Pages[tabName] = PageScroll

    local TabBtn = Instance.new("TextButton")
    TabBtn.Size = UDim2.new(1, 0, 0, 38)
    TabBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 36)
    TabBtn.Text = tabName
    TabBtn.TextColor3 = Color3.fromRGB(210, 210, 210)
    TabBtn.Font = Enum.Font.GothamMedium
    TabBtn.TextSize = 13
    TabBtn.Parent = Sidebar

    table.insert(activeTabButtons, TabBtn)

    local TabCorner = Instance.new("UICorner")
    TabCorner.CornerRadius = UDim.new(0, 6)
    TabCorner.Parent = TabBtn

    TabBtn.MouseButton1Click:Connect(function()
        TweenService:Create(TabBtn, TweenInfo.new(0.1), {Size = UDim2.new(0.95, 0, 0, 36)}):Play()
        task.wait(0.1)
        TweenService:Create(TabBtn, TweenInfo.new(0.1), {Size = UDim2.new(1, 0, 0, 38)}):Play()

        for _, p in pairs(Pages) do p.Visible = false end
        for _, b in ipairs(Sidebar:GetChildren()) do
            if b:IsA("TextButton") then
                b.BackgroundColor3 = Color3.fromRGB(28, 28, 36)
                b.TextColor3 = Color3.fromRGB(210, 210, 210)
            end
        end
        PageScroll.Visible = true
        TabBtn.BackgroundColor3 = MainStroke.Color
        TabBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
        currentActivePage = tabName
    end)

    if not currentActivePage then
        PageScroll.Visible = true
        TabBtn.BackgroundColor3 = Color3.fromRGB(255, 190, 0)
        TabBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
        currentActivePage = tabName
    end

    return PageScroll
end

local CharTab = CreatePage("اللاعب | Character")

local CustomSpeed = 16
local CustomJumpPower = 50
local SpeedEnabled = false
local JumpEnabled = false
local InfiniteJumpEnabled = false
local NoclipEnabled = false
local SitFlyEnabled = false

local sitAnimTrack = nil
RunService.RenderStepped:Connect(function()
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        
        if hum then
            if SpeedEnabled and hum.WalkSpeed ~= CustomSpeed then
                hum.WalkSpeed = CustomSpeed
            end
            if JumpEnabled then
                hum.UseJumpPower = true
                if hum.JumpPower ~= CustomJumpPower then
                    hum.JumpPower = CustomJumpPower
                end
            end
        end

        if NoclipEnabled then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end

        if SitFlyEnabled and rootPart and hum then
            hum.PlatformStand = true
            if not sitAnimTrack then
                local anim = Instance.new("Animation")
                anim.AnimationId = "rbxassetid://2506281703"
                sitAnimTrack = hum:LoadAnimation(anim)
                sitAnimTrack:Play()
            end
            local cam = workspace.CurrentCamera
            local moveDir = Vector3.new(0, 0, 0)
            local flySpeed = CustomSpeed * 2

            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

            rootPart.Velocity = moveDir * flySpeed
            rootPart.CFrame = CFrame.new(rootPart.Position, rootPart.Position + cam.CFrame.LookVector)
        else
            if hum and hum.PlatformStand and not SitFlyEnabled then
                hum.PlatformStand = false
                if sitAnimTrack then
                    sitAnimTrack:Stop()
                    sitAnimTrack = nil
                end
            end
        end
    end)
end)

local function AddSlider(page, name, minVal, maxVal, defaultVal, callback)
    local Frame = Instance.new("Frame")
    Frame.Size = UDim2.new(1, -5, 0, 50)
    Frame.BackgroundColor3 = Color3.fromRGB(26, 26, 36)
    Frame.Parent = page
    Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 8)

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(0.6, 0, 0.4, 0)
    Label.Position = UDim2.new(0, 12, 0.1, 0)
    Label.BackgroundTransparency = 1
    Label.Text = name
    Label.TextColor3 = Color3.fromRGB(220, 220, 240)
    Label.Font = Enum.Font.GothamMedium
    Label.TextSize = 13
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Frame

    local ValLabel = Instance.new("TextLabel")
    ValLabel.Size = UDim2.new(0.3, 0, 0.4, 0)
    ValLabel.Position = UDim2.new(0.65, 0, 0.1, 0)
    ValLabel.BackgroundTransparency = 1
    ValLabel.Text = tostring(defaultVal)
    ValLabel.TextColor3 = Color3.fromRGB(255, 190, 0)
    ValLabel.Font = Enum.Font.GothamBold
    ValLabel.TextSize = 13
    ValLabel.TextXAlignment = Enum.TextXAlignment.Right
    ValLabel.Parent = Frame

    local SliderBar = Instance.new("Frame")
    SliderBar.Size = UDim2.new(1, -24, 0, 6)
    SliderBar.Position = UDim2.new(0, 12, 0.65, 0)
    SliderBar.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
    SliderBar.Parent = Frame
    Instance.new("UICorner", SliderBar).CornerRadius = UDim.new(1, 0)

    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new((defaultVal - minVal)/(maxVal - minVal), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(255, 190, 0)
    Fill.Parent = SliderBar
    Instance.new("UICorner", Fill).CornerRadius = UDim.new(1, 0)

    local DraggingBar = false
    local function Update(input)
        local pos = math.clamp((input.Position.X - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
        local value = math.floor(minVal + (maxVal - minVal) * pos)
        Fill.Size = UDim2.new(pos, 0, 1, 0)
        ValLabel.Text = tostring(value)
        callback(value)
    end

    SliderBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            DraggingBar = true
            Update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            DraggingBar = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if DraggingBar and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            Update(input)
        end
    end)
end

                bangConn = RunService.Heartbeat:Connect(function()
                    pcall(function()
                        if target.Character and target.Character:FindFirstChild("HumanoidRootPart") and root then
                            local tRoot = target.Character.HumanoidRootPart
                            local wave = math.sin(tick() * 30) * 1.2
                            root.CFrame = tRoot.CFrame * CFrame.new(0, 0, 1.5 + wave)
                            root.Velocity = Vector3.new(0, 0, 0)
                        end
                    end)
                end)
            end
        end)
    else
        btn.BackgroundColor3 = Color3.fromRGB(160, 25, 25)
        if bangConn then bangConn:Disconnect() bangConn = nil end
        pcall(function()
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum then hum.PlatformStand = false end
        end)
    end
end)

-- =================================================================
-- أنميشن النزول من أعلى الشاشة إلى المكان الحالي (إظهار/إخفاء بزر K)
-- =================================================================
MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
local targetCustomPosition = MainFrame.Position
local IsOpen = false

local function ToggleUI()
    IsOpen = not IsOpen
    
    if IsOpen then
        targetCustomPosition = MainFrame.Position
        MainFrame.Position = UDim2.new(targetCustomPosition.X.Scale, targetCustomPosition.X.Offset, 0, -450)
        MainFrame.Visible = true
        
        TweenService:Create(MainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Position = targetCustomPosition
        }):Play()
    else
        local tw = TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Position = UDim2.new(MainFrame.Position.X.Scale, MainFrame.Position.X.Offset, 0, -450)
        })
        tw:Play()
        tw.Completed:Connect(function()
            if not IsOpen then
                MainFrame.Visible = false
                MainFrame.Position = targetCustomPosition
            end
        end)
    end
end

task.spawn(function()
    task.wait(0.1)
    ToggleUI()
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.K then
        ToggleUI()
    end
end)
